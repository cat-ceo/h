%macro WRITE 2
    mov rax, 01
    mov rdi, 01
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro

%macro READ 2
    mov rax, 00
    mov rdi, 00
    mov rsi, %1
    mov rdx, %2
    syscall
%endmacro

%macro EXIT 0
    mov rax, 60
    mov rdi, 00
    syscall
%endmacro

section .data
    msg1 db "Enter the BCD number: "
    len1 equ $-msg1
    msg2 db "This is the hexadecimal no.: "
    len2 equ $-msg2

section .bss
    char_buff resb 32
    act1 resq 1
    num resq 1

section .text
    global _start

_start:
    WRITE msg1, len1
    READ char_buff, 17
    dec rax
    mov qword[act1], rax

    mov rsi, char_buff
    mov rax, 00H
    mov rbx, 0AH

up1:
    mov rdx, 00H
    mov dl, byte[rsi]
    sub dl, 30H
    mov rcx, rdx
    mul rbx
    add rax, rcx
    inc rsi
    dec qword[act1]
    jnz up1

    mov qword[num], rax

    WRITE msg2, len2
    mov rbx, qword[num]
    call display_hex

    EXIT

display_hex:
    mov rsi, char_buff
    mov rcx, 16

above:
    rol rbx, 04
    mov dl, bl
    and dl, 0FH
    cmp dl, 09H
    jbe add30
    add dl, 07H

add30:
    add dl, 30H
    mov byte[rsi], dl
    inc rsi
    dec rcx
    jnz above

    WRITE char_buff, 16
    ret
