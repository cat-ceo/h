; ------------------------------------
; 2. String copy
; ------------------------------------
strcpy:
    mov rsi, string1
    mov rdi, string3
    mov rcx, [l1]
    cld
    rep movsb

    write msg4, len4
    write string3, [l1]
    write newline, 1
    jmp printmenu

; ------------------------------------
; 3. String reverse
; ------------------------------------
strrev:
    mov rcx, [l1]
    test rcx, rcx
    jz reverse_done

    mov rsi, string1
    add rsi, rcx
    dec rsi
    mov rdi, string3

reverse_loop:
    mov al, [rsi]
    mov [rdi], al
    dec rsi
    inc rdi
    dec rcx
    jnz reverse_loop

reverse_done:
    write msg5, len5
    write string3, [l1]
    write newline, 1
    jmp printmenu

; ------------------------------------
; 4. String compare
; ------------------------------------
strcmp:
    mov rax, [l1]
    cmp rax, [l2]
    jne not_equal

    mov rsi, string1
    mov rdi, string2
    mov rcx, [l1]
    cld
    repe cmpsb
    jne not_equal

    write msg6, len6
    jmp printmenu

not_equal:
    write msg7, len7
    jmp printmenu

; ------------------------------------
; 5. String concatenation
; ------------------------------------
concat:
    mov rsi, string1
    mov rdi, string3
    mov rcx, [l1]
    cld
    rep movsb

    mov rsi, string2
    mov rcx, [l2]
    rep movsb

    mov rax, [l1]
    add rax, [l2]
    mov [l3], rax

    write msg8, len8
    write string3, [l3]
    write newline, 1
    jmp printmenu

; ------------------------------------
; 6. Palindrome check
; ------------------------------------
strpalindrome:
    mov rsi, string1
    mov rdi, string3
    mov rcx, [l1]

    test rcx, rcx
    jz palindrome_yes

    add rsi, rcx
    dec rsi

palindrome_reverse:
    mov al, [rsi]
    mov [rdi], al
    dec rsi
    inc rdi
    dec rcx
    jnz palindrome_reverse

    mov rsi, string1
    mov rdi, string3
    mov rcx, [l1]
    cld
    repe cmpsb
    jne palindrome_no

palindrome_yes:
    write msg9, len9
    jmp printmenu

palindrome_no:
    write msg10, len10
    jmp printmenu

; ------------------------------------
; Read a line and remove its newline
; Input: RSI = buffer, RDX = max bytes
; Output: RAX = string length
; ------------------------------------
read_line:
    push rbx
    mov rbx, rsi
    xor rax, rax
    xor rdi, rdi
    syscall

    test rax, rax
    jle read_empty

    xor rcx, rcx

read_scan:
    cmp rcx, rax
    jae read_finish
    cmp byte [rbx + rcx], 10
    je read_newline
    inc rcx
    jmp read_scan

read_newline:
    mov byte [rbx + rcx], 0
    mov rax, rcx
    jmp read_return

read_finish:
    mov byte [rbx + rax], 0
    jmp read_return

read_empty:
    xor rax, rax

read_return:
    pop rbx
    ret

; ------------------------------------
; Display unsigned decimal integer
; Input: RAX = number
; ------------------------------------
display:
    lea rsi, [buff + 23]
    mov byte [rsi], 10
    mov rcx, 1
    mov rbx, 10
    test rax, rax
    jnz display_loop

    dec rsi
    mov byte [rsi], '0'
    inc rcx
    jmp display_write

display_loop:
    xor rdx, rdx
    div rbx
    add dl, '0'
    dec rsi
    mov [rsi], dl
    inc rcx
    test rax, rax
    jnz display_loop

display_write:
    mov rdx, rcx
    mov rax, 1
    mov rdi, 1
    syscall
    ret

; ------------------------------------
; 7. Exit
; ------------------------------------
exit:
    mov rax, 60
    xor rdi, rdi
    syscall

    mov rax,qword[a]

    call display

    mov rdx,00H

    ; Quotient is in RAX
    ; Remainder is in RDX

    mov qword[c],rax

    WRITE msg_div,len_div

    mov rbx,qword[c]
    call display

    jmp printmenu

    jmp printmenu

; SUCCESSIVE ADDITION METHOD
succadd:
    mov rbx,00
    mov rdx,00
    mov rcx,[n]

    test rcx,rcx
    jz succdone

l1:
    add rbx,[m]
    adc rdx,00
    dec rcx
    jnz l1

succdone:
    mov [ans],rbx
    mov [A],rdx
    WRITE msg4,len4
    mov rdx,[A]
    mov rbx,[ans]
    call display
    jmp printmenu

; SHIFT AND ADD METHOD
shiftadd:
    mov qword [A],00H
    mov rbx,[m]
    mov [B],rbx
    mov rbx,[n]
    mov [Q],rbx
    mov rcx,64

l2:
    mov rbx,[Q]
    and rbx,01H
    jz shiftaq
    mov rbx,[B]
    add [A],rbx

shiftaq:
    mov rbx,[Q]
    shr rbx,01H
    mov rdx,[A]
    and rdx,01H
    shl rdx,63
    or rbx,rdx
    mov [Q],rbx
    shr qword [A],01H
    dec rcx
    jnz l2

    WRITE msg4,len4
    mov rdx,[A]
    mov rbx,[Q]
    call display
    jmp printmenu

exit:
    EXIT

    add r11b,07H

    add r11b,30H

extern printf,scanf

%macro PRINT 02

mov rdi,%1
mov rsi,%2
call printf

%endmacro

%macro SCAN 02

mov rdi,%1
mov rsi,%2
call scanf

%endmacro

%endmacro
