%macro WRITE 2
 mov rax, 01
 mov rdi, 01
 mov rsi, %1
 mov rdx, %2
 syscall
%endmacro
%macro READ 2
 mov rax, 00
 mov rdi, 00
 mov rsi, %1
 mov rdx, %2
 syscall
%endmacro
%macro EXIT 0
 mov rax, 60
 mov rdi, 00
 syscall
%endmacro
section .data
menu db 10,"1. Addition",10
 db "2. Subtraction",10
 db "3. Multiplication",10
 db "4. Division",10
 db "5. Exit",10
 db "Enter your choice: ",10
menulen equ $-menu
msg1 db "Enter 2 numbers",10
len1 equ $-msg1
msg_add db "Addition is ",10
len_add equ $-msg_add
msg_sub db "Subtraction is ",10
len_sub equ $-msg_sub
msg_mul db "Multiplication is ",10
len_mul equ $-msg_mul
msg_div db "Division is ",10
len_div equ $-msg_div
msg_zero db "Cannot divide by zero!",10
len_zero equ $-msg_zero
msg_invalid db "Invalid choice!",10
len_invalid equ $-msg_invalid
section .bss
char_buff resb 17
actl resq 1
choice resb 2
a resq 1
b resq 1
c resq 1
section .text
global _start
_start:
 ; -------------------------
 ; Accept two numbers
 ; -------------------------
 WRITE msg1,len1
 call accept
 mov qword[a],rbx
 call accept
 mov qword[b],rbx
; ==================================
; MENU
; ==================================
printmenu:
 WRITE menu,menulen
 READ choice,2
 cmp byte[choice],31H
 je addition
 cmp byte[choice],32H
 je subtraction
 cmp byte[choice],33H
 je multiplication
 cmp byte[choice],34H
 je division
 cmp byte[choice],35H
 je exit
 ; Invalid choice
 WRITE msg_invalid,len_invalid
 jmp printmenu
; ==================================
; ADDITION
; ==================================
addition:
 mov rbx,qword[a]
 add rbx,qword[b]
 mov qword[c],rbx
 WRITE msg_add,len_add
 mov rbx,qword[c]
 call display
 jmp printmenu
; ==================================
; SUBTRACTION
; ==================================
subtraction:
 mov rbx,qword[a]
 sub rbx,qword[b]
 mov qword[c],rbx
 WRITE msg_sub,len_sub
 mov rbx,qword[c]
 call display
 jmp printmenu
; ==================================
; MULTIPLICATION
; ==================================
multiplication:
 mov rax,qword[a]
 mul qword[b]
 ; RDX:RAX contains result
 ; We store lower 64 bits in c
 mov qword[c],rax
 WRITE msg_mul,len_mul
 mov rbx,qword[c]
 call display
 jmp printmenu
; ==================================
; DIVISION
; ==================================
division:
 cmp qword[b],00H
 je zero_error
 mov rax,qword[a]
 mov rdx,00H
 div qword[b]
 ; Quotient is in RAX
 ; Remainder is in RDX
 mov qword[c],rax
 WRITE msg_div,len_div
 mov rbx,qword[c]
 call display
 jmp printmenu
zero_error:
 WRITE msg_zero,len_zero
 jmp printmenu
; ==================================
; EXIT
; ==================================
exit:
 EXIT
; ==================================
; ACCEPT
; ==================================
accept:
 READ char_buff,17
 dec rax
 mov qword[actl],rax
 mov rsi,char_buff
 mov rbx,00H
up:
 shl rbx,04H
 mov rdx,00H
 mov dl,byte[rsi]
 cmp dl,39H
 jbe sub30
 sub dl,07H
sub30:
 sub dl,30H
 add rbx,rdx
 inc rsi
 dec qword[actl]
 jnz up
 RET
; ==================================
; DISPLAY
; ==================================
display:
 mov rsi,char_buff
 mov rcx,16
above:
 rol rbx,04H
 mov dl,bl
 AND dl,0FH
 cmp dl,09H
 jbe add30
 add dl,07H
add30:
 add dl,30H
 mov byte[rsi],dl
 inc rsi
 dec rcx
 jnz above
 WRITE char_buff,16
 RET
