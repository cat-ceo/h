%macro WRITE 2
 mov rax, 01
 mov rdi, 01
 mov rsi, %1
 mov rdx, %2
 syscall
%endmacro
%macro READ 2
 mov rax, 00
 mov rdi, 00
 mov rsi, %1
 mov rdx, %2
 syscall
%endmacro
%macro EXIT 0
 mov rax, 60
 mov rdi, 00
 syscall
%endmacro
section .data
menu db 10,"1. BCD to Hex",10
 db "2. Hex to BCD",10
 db "3. Exit",10
 db "Enter your choice: "
menulen equ $-menu
msg1 db "Enter the BCD number: "
len1 equ $-msg1
msg2 db "This is the hexadecimal no.: "
len2 equ $-msg2
msg3 db "Enter the Hexadecimal number: "
len3 equ $-msg3
msg4 db "The BCD number is: "
len4 equ $-msg4
msg5 db "Invalid choice!",10
len5 equ $-msg5
section .bss
char_buff resb 32
act1 resq 1
num resq 1
choice resb 1
section .text
global _start
_start:
 WRITE menu,menulen
 READ choice,02
 cmp byte[choice],31H
 je bcd2hex
 cmp byte[choice],32H
 je hex2bcd
 cmp byte[choice],33H
 je exit
 WRITE msg5,len5
 jmp _start
bcd2hex:
 WRITE msg1,len1
 READ char_buff,17
 dec rax
 mov qword[act1],rax
 mov rsi,char_buff
 mov rax,00H
 mov rbx,0AH
up1:
 mov rdx,00H
 mov dl,byte[rsi]
 sub dl,30H
 mov rcx,rdx
 mul rbx
 add rax,rcx
 inc rsi
 dec qword[act1]
 jnz up1
 mov qword[num],rax
 WRITE msg2,len2
 mov rbx,qword[num]
 call display_hex
 jmp _start
hex2bcd:
 WRITE msg3,len3
 READ char_buff,17
 dec rax
 mov qword[act1],rax
 mov rsi,char_buff
 mov rax,00H
 mov rbx,10H
up2:
 mov rdx,00H
 mov dl,byte[rsi]
 cmp dl,39H
 jbe digit
 sub dl,07H
digit:
 sub dl,30H
 mov rcx,rdx
 mul rbx
 add rax,rcx
 inc rsi
 dec qword[act1]
 jnz up2
 mov qword[num],rax
 WRITE msg4,len4
 call display_bcd
 jmp _start
display_hex:
 mov rsi,char_buff
 mov rcx,16
above:
 rol rbx,04
 mov dl,bl
 and dl,0FH
 cmp dl,09H
 jbe add30
 add dl,07H
add30:
 add dl,30H
 mov byte[rsi],dl
 inc rsi
 dec rcx
 jnz above
 WRITE char_buff,16
 ret
display_bcd:
 mov rax,qword[num]
 mov rsi,char_buff
 add rsi,31
 mov rcx,00H
convert:
 mov rdx,00H
 mov rbx,10
 div rbx
 add dl,30H
 dec rsi
 mov byte[rsi],dl
 inc rcx
 cmp rax,00H
 jnz convert
 mov rdx,rcx
 WRITE rsi,rdx
 ret
exit:
 EXIT
